   234→# ============== COOKIE EXTRACTION LOGIC ==============
   235→def extract_cookie_from_message(message_text: str, cookie_type: str) -> Optional[str]:
   236→    """
   237→    Extract cookie/data from message based on type
   238→    Different patterns for different services
   239→    """
   240→    if not message_text:
   241→        return None
   242→    
   243→    text = message_text.strip()
   244→    
   245→    # Patterns based on cookie type
   246→    patterns = {
   247→        "netflix": [
   248→            # Full Netflix URL with nftoken (PRIMARY - this is what we want!)
   249→            r'(https://www\.netflix\.com/browse\?nftoken=[^\s"\'<>]+)',
   250→            # Alternative: nftoken in any URL format
   251→            r'(https?://[^\s]*nftoken=[^\s]+)',
   252→            # Fallback: just the nftoken value if URL not present
   253→            r'(nftoken=([A-Za-z0-9+/=]{100,}))',
   254→        ],
   255→        "hotstar": [
   256→            r'(st=[^;]+)',
   257→            r'(hotstar[^;]*;[^;]*)',
   258→            r'([a-zA-Z0-9]{30,})',
   259→        ],
   260→        "prime": [
   261→            r'(prime-[a-zA-Z0-9.-]+)',
   262→            r'(at-main[\s\S]*?at-cookie)',
   263→            r'([a-zA-Z0-9]{40,})',
   264→        ],
   265→        "crunchyroll": [
   266→            r'(etp_rt=[^;]+)',
   267→            r'(session_id=[^;]+)',
   268→            r'([a-zA-Z0-9]{30,})',
   269→        ]
   270→    }
   271→    
   272→    # Generic fallback pattern
   273→    generic_patterns = [
   274→        r'```([^`]+)```',  # Code blocks
   275→        r'([a-zA-Z0-9+/=]{50,})',  # Long encoded strings
   276→        r'(\{[^{}]+\})',  # JSON-like
   277→    ]
   278→    
   279→    # Try type-specific patterns first
   280→    for pattern in patterns.get(cookie_type, []):
   281→        match = re.search(pattern, text, re.IGNORECASE | re.DOTALL)
   282→        if match:
   283→            result = match.group(1).strip()
   284→            # For nftoken pattern, return full "nftoken=value" or just value if captured in group 2
   285→            if match.lastindex and match.lastindex >= 2:
   286→                result = match.group(2).strip()  # Return the actual token value
   287→            return result
   288→    
   289→    # Fallback to generic patterns
   290→    for pattern in generic_patterns:
   291→        match = re.search(pattern, text, re.IGNORECASE | re.DOTALL)
   292→        if match:
   293→            return match.group(1).strip()
   294→    
   295→    # If no pattern matches but message is long enough, return it
   296→    if len(text) > 30 and not text.startswith('/'):
   297→        return text[:500]  # Limit length
   298→    
   299→    return None

   302→# ============== COLLECTION ENGINE ==============
   303→async def run_collection_job(job_id: str):
   304→    """Background task that runs the actual collection"""
   305→    global job_counter
   306→    
   307→    job = active_jobs.get(job_id)
   308→    if not job:
   309→        return
   310→    
   311→    job.status = "running"
   312→    job.started_at = datetime.now().isoformat()
   313→    
   314→    tg_client = await get_telegram_client()
   315→    
   316→    if not tg_client:
   317→        job.status = "error"
   318→        job.error = "Telegram client not available. Check API credentials."
   319→        job.completed_at = datetime.now().isoformat()
   320→        return
   321→    
   322→    # Track consecutive failures for rate limit protection
   323→    consecutive_failures = 0
   324→    max_consecutive_failures = 10  # Stop after 10 consecutive failures
   325→    
   326→    # CRITICAL FIX: Track message IDs to avoid reading same cookie twice!
   327→    last_processed_msg_id = 0
   328→    button_failures = 0  # Track how many times button click fails
   329→    max_button_failures = 3  # Reset conversation after this many failures
   330→    
   331→    try:
   332→        # Get target bot entity
   333→        target_bot = await tg_client.get_entity(TARGET_BOT)
   334→        print(f"✅ Connected to target bot: @{target_bot.username}")
   335→        
   336→        # Send /start to initiate conversation
   337→        await tg_client.send_message(target_bot, "/start")
   338→        print("📤 Sent /start command")
   339→        await asyncio.sleep(3)  # Wait for response
   340→        
   341→        # Get initial message ID as starting point (don't process old messages!)
   342→        initial_messages = []
   343→        async for msg in tg_client.iter_messages(target_bot, limit=1):
   344→            initial_messages.append(msg)
   345→        if initial_messages:
   346→            last_processed_msg_id = initial_messages[0].id
   347→            print(f"📍 Starting from message ID: {last_processed_msg_id} (will only read newer)")
   348→        
   349→        # Get cookie type info for button matching
   350→        type_info = COOKIE_TYPES.get(job.cookie_type, {})
   351→        
   352→        while job.collected_count < job.target_count and job.status == "running":
   353→            print(f"\n--- Attempting cookie #{job.collected_count + 1} ---")
   354→            
   355→            # Step 1: Click the NF Cookie / Service button FIRST
   356→            print("🖱️ Looking for cookie button to click...")
   357→            button_clicked = await click_cookie_button(tg_client, target_bot, job.cookie_type)
   358→            
   359→            if button_clicked:
   360→                print("✅ Button clicked! Waiting 15 seconds for new cookie...")
   361→                await asyncio.sleep(15)  # Wait for bot to generate and send cookie
   362→                button_failures = 0  # Reset on success
   363→            else:
   364→                button_failures += 1
   365→                print(f"⚠️ No button found (failure #{button_failures}/{max_button_failures})")
   366→                
   367→                if button_failures >= max_button_failures:
   368→                    print("🔄 Too many button failures, resetting conversation with /start...")
   369→                    await tg_client.send_message(target_bot, "/start")
   370→                    await asyncio.sleep(5)
   371→                    button_failures = 0
   372→                
   373→                await asyncio.sleep(5)
   374→            
   375→            # Step 2: Read ONLY messages NEWER than last_processed_msg_id
   376→            # This is the KEY FIX - prevents reading same cookie twice!
   377→            messages = []
   378→            
   379→            if last_processed_msg_id > 0:
   380→                # Get messages with ID greater than last processed
   381→                async for msg in tg_client.iter_messages(
   382→                    target_bot, 
   383→                    limit=10,
   384→                    min_id=last_processed_msg_id  # ONLY get newer messages!
   385→                ):
   386→                    messages.append(msg)
   387→            else:
   388→                # Fallback: just get latest
   389→                async for msg in tg_client.iter_messages(target_bot, limit=3):
   390→                    messages.append(msg)
   391→            
   392→            if not messages:
   393→                print("⏳ No NEW messages yet, waiting...")
   394→                consecutive_failures += 1
   395→                await asyncio.sleep(3)
   396→                
   397→                if consecutive_failures >= max_consecutive_failures:
   398→                    print(f"❌ Stopping: {consecutive_failures} consecutive failures (no new messages)")
   399→                    job.status = "error"
   400→                    job.error = "Bot stopped responding. No new messages received."
   401→                    job.completed_at = datetime.now().isoformat()
   402→                    return
   403→                continue
   404→            
   405→            # Sort by ID ascending to get the OLDEST new message first
   406→            messages.sort(key=lambda m: m.id)
   407→            latest_msg = messages[-1]  # Get the newest of the new messages
   408→            
   409→            print(f"📩 Reading NEW message ID {latest_msg.id} (previous was {last_processed_msg_id}): {(latest_msg.text or '')[:50]}...")
   410→            
   411→            # Update tracking BEFORE processing (prevent re-reading on error)
   412→            last_processed_msg_id = max(last_processed_msg_id, latest_msg.id)
   413→            
   414→            # Step 3: Extract cookie from message
   415→            cookie_data = extract_cookie_from_message(latest_msg.text or "", job.cookie_type)
   416→            
   417→            if cookie_data:
   418→                # Check for duplicates before saving
   419→                is_duplicate = False
   420→                for existing_cookie in all_cookies:
   421→                    if existing_cookie.data == cookie_data:
   422→                        is_duplicate = True
   423→                        break
   424→                
   425→                if not is_duplicate:
   426→                    # Save the cookie
   427→                    job.collected_count += 1
   428→                    new_cookie = CollectedCookie(
   429→                        id=len(all_cookies) + 1,
   430→                        cookie_type=job.cookie_type,
   431→                        data=cookie_data,
   432→                        timestamp=datetime.now().isoformat(),
   433→                        raw_message=(latest_msg.text or "")[:300]
   434→                    )
   435→                    job.cookies.append(new_cookie)
   436→                    all_cookies.append(new_cookie)
   437→                    
   438→                    print(f"✅ Cookie #{job.collected_count}/{job.target_count} SAVED!")
   439→                    print(f"   Preview: {cookie_data[:70]}..." )
   440→                    
   441→                    # Save to file
   442→                    save_cookie_to_file(new_cookie)
   443→                    
   444→                    # Reset failure counter on success
   445→                    consecutive_failures = 0
   446→                else:
   447→                    print(f"⚠️ Duplicate cookie detected, skipping...")
   448→                    consecutive_failures += 1
   449→                    
   450→                # Small delay between successful collections to avoid rate limiting
   451→                await asyncio.sleep(3)
   452→            else:
   453→                print(f"ℹ️ No cookie in this message, waiting before retry...")
   454→                consecutive_failures += 1
   455→                # CRITICAL FIX: Wait before retrying to avoid rate limiting!
   456→                await asyncio.sleep(5)
   457→            
   458→            # Check if we've hit max consecutive failures
   459→            if consecutive_failures >= max_consecutive_failures:
   460→                print(f"❌ Stopping job: {consecutive_failures} consecutive failures")
   461→                job.status = "error"
   462→                job.error = f"Too many consecutive failures ({max_consecutive_failures}). Bot may be rate limited or offline."
   463→                job.completed_at = datetime.now().isoformat()
   464→                return
   465→        
   466→        # Mark as completed
   467→        job.status = "completed"
   468→        job.completed_at = datetime.now().isoformat()
   469→        print(f"\n🎉 Job {job_id} completed! Total cookies: {job.collected_count}")
   470→        
   471→    except Exception as e:
   472→        print(f"❌ Job failed: {e}")
   473→        import traceback
   474→        traceback.print_exc()
   475→        job.status = "error"
   476→        job.error = str(e)
   477→        job.completed_at = datetime.now().isoformat()

   480→async def click_cookie_button(tg_client, target_bot, cookie_type: str) -> bool:
   481→    """
   482→    Click the cookie generation button (e.g., 'NF Token', 'NF Cookie', etc.)
   483→    This is the MAIN button that generates cookies - NOT a 'next' button!
   484→    Returns True if clicked successfully.
   485→    """
   486→    try:
   487→        # Get latest message with buttons
   488→        messages = []
   489→        async for msg in tg_client.iter_messages(target_bot, limit=5):
   490→            if msg.buttons:
   491→                messages.append(msg)
   492→        
   493→        if not messages:
   494→            print("⚠️ No messages with buttons found")
   495→            return False
   496→        
   497→        # Try multiple messages (not just latest) in case buttons expired
   498→        for msg_idx, msg_with_buttons in enumerate(messages):
   499→            print(f"   Checking message {msg_idx + 1} for buttons...")
   500→            
   501→            # Search for NF Token / Cookie button with MULTIPLE patterns
   502→            button_patterns = [
   503→                "nf token", "nf cookie", "nf", "netflix",
   504→                "cookie", "token", "get", "generate"
   505→            ]
   506→            
   507→            for row in msg_with_buttons.buttons:
   508→                for btn in row:
   509→                    btn_text = btn.text.lower().strip()
   510→                    print(f"      Found button: '{btn.text}'")
   511→                    
   512→                    # Check if this button matches any pattern
   513→                    is_target_button = False
   514→                    for pattern in button_patterns:
   515→                        if pattern in btn_text:
   516→                            is_target_button = True
   517→                            break
   518→                    
   519→                    if is_target_button:
   520→                        print(f"🖱️ CLICKING BUTTON: '{btn.text}' ✅")
   521→                        await btn.click()
   522→                        return True
   523→        
   524→        # If we get here, no matching button was found
   525→        print("⚠️ No suitable cookie button found!")
   526→        print("   Available buttons:")
   527→        for msg_with_buttons in messages[:2]:  # Show first 2 messages
   528→            for row in msg_with_buttons.buttons:
   529→                for btn in row:
   530→                    print(f"      - '{btn.text}'")
   531→        
   532→        return False
   533→        
   534→    except Exception as e:
   535→        print(f"❌ Error clicking button: {e}")
   536→        return False

   624→@app.post("/api/collect")
   625→async def start_collection(
   626→    background_tasks: BackgroundTasks,
   627→    cookie_type: str = Query(..., description="Cookie type: netflix, hotstar, prime, crunchyroll"),
   628→    count: int = Query(..., ge=1, le=1000, description="Number of cookies to collect (1-1000)")
   629→):
   630→    """Start a new collection job"""
   631→    global job_counter
   632→    job_counter += 1
   633→    
   634→    # Validate cookie type
   635→    if cookie_type not in COOKIE_TYPES:
   636→        raise HTTPException(status_code=400, detail={
   637→            "error": f"Invalid cookie type. Choose from: {list(COOKIE_TYPES.keys())}"
   638→        })
   639→    
   640→    # Create job
   641→    job_id = f"job_{job_counter}_{datetime.now().strftime('%H%M%S')}"
   642→    
   643→    job = CollectionJob(
   644→        id=job_id,
   645→        cookie_type=cookie_type,
   646→        target_count=count,
   647→        status="pending"
   648→    )
   649→    
   650→    active_jobs[job_id] = job
   651→    
   652→    # Start background collection
   653→    background_tasks.add_task(run_collection_job, job_id)
   654→    
   655→    return {
   656→        "success": True,
   657→        "job_id": job_id,
   658→        "message": f"Collection started for {COOKIE_TYPES[cookie_type]['name']}",
   659→        "target_count": count,
   660→        "status_url": f"/api/job/{job_id}",
   661→        "websocket_hint": "Poll this URL for updates every 2 seconds"
   662→    }

   699→    
   700→    return {
   701→        "success": True,
   702→        "message": "Job stopped",
   703→        "collected_so_far": job.collected_count

   707→@app.get("/api/cookies")
   708→async def get_all_cookies(
   709→    cookie_type: Optional[str] = None,
   710→    limit: int = Query(100, ge=1, le=1000),
   711→    offset: int = Query(0, ge=0)
   712→):
   713→    """Get all collected cookies with filtering"""
   714→    cookies = all_cookies
   715→    
   716→    # Filter by type
   717→    if cookie_type and cookie_type in COOKIE_TYPES:
   718→        cookies = [c for c in cookies if c.cookie_type == cookie_type]
   719→    
   720→    # Paginate
   721→    total = len(cookies)
   722→    cookies_page = cookies[offset:offset+limit]
   723→    
   724→    return {
   725→        "cookies": [c.to_dict() for c in cookies_page],
   726→        "total": total,
   727→        "showing": len(cookies_page),
   728→        "offset": offset,
   729→        "limit": limit
   730→    }
